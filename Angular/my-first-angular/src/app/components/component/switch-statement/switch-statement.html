Switch Statement

Basic Switch

@switch (day) { @case ('monday') {

Start of the week

} @case ('friday') {

Last working day

} @case ('sunday') {

Holiday

} @default {

Normal day

} }

Switch with Button

@switch (color) { @case ('red') {

Red is selected

} @case ('green') {

Green is selected

} @case ('blue') {

Blue is selected

} @default {

No color selected

} }

Switch with Input

@switch (role) { @case ('admin') {

Admin can do everything

} @case ('user') {

User can only read

} @case ('guest') {

Guest has no access

} @default {

Enter a valid role

} }
